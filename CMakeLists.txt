cmake_minimum_required(VERSION 3.25)
project(EndfieldHudFields VERSION 0.1.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(HudFields SHARED src/plugin.cpp)
target_include_directories(HudFields PRIVATE src sdk/include)
set_target_properties(HudFields PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/hud-fields")
add_custom_target(package ALL DEPENDS HudFields)
foreach(file mod.ini config.ini hud-fields.lua icon.png zml-package.json)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_CURRENT_SOURCE_DIR}/mod/${file}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/hud-fields/${file}")
endforeach()
enable_testing()
add_executable(PatchTests tests/patch_tests.cpp)
target_include_directories(PatchTests PRIVATE src)
add_test(NAME patch COMMAND PatchTests)
add_executable(PluginTests tests/plugin_tests.cpp)
target_include_directories(PluginTests PRIVATE sdk/include)
add_test(NAME plugin COMMAND PluginTests "$<TARGET_FILE:HudFields>")
set(ZML_LUPA_PATH "" CACHE PATH "Optional installed Lupa path")
set(ZML_SERVICES_TEST_EXE "" CACHE FILEPATH "Optional external loader services fixture")
if(ZML_LUPA_PATH AND ZML_SERVICES_TEST_EXE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME lua_behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/lua_tests.py"
    --lupa-dir "${ZML_LUPA_PATH}" --services "${ZML_SERVICES_TEST_EXE}")
  set_tests_properties(lua_behavior PROPERTIES TIMEOUT 30)
endif()
